    package main

import "fmt"

type PessoaFila struct {
	nome			string
	distancia	int
}

func main() {
	conexoes := map[string][]string{
    "Ana":      {"Bruno", "Carlos"},
    "Bruno":    {"Ana", "Daniel", "Eduardo"},
    "Carlos":   {"Ana", "Fernanda"},
    "Daniel":   {"Bruno"},
    "Eduardo":  {"Bruno"},
    "Fernanda": {"Carlos"},
	}

	fila := []PessoaFila{
		{"Ana", 0},
	}

	pontoFinal := "Daniel"
	distancia := 0

	for len(fila) > 0 {
			pessoa := fila[0]
			distancia++

			if pessoa.nome == pontoFinal {
				fmt.Printf("%s está a %d passos de distância", pessoa.nome, pessoa.distancia)
				break
			}
			for _, pessoaConectada := range conexoes[pessoa.nome] {
				fila = append(fila, PessoaFila{
					nome: pessoaConectada,
					distancia: distancia,
				})	
			}
			fila = fila[1:]
		}
}
